```typescript
import { useQuery } from "@tanstack/react-query";
import type { PaginatedResponse, ProviderSearchResult, ProviderProfile, Availability } from "@mysitter/types";
import { apiClient } from "../lib/apiClient";

export function useProviderSearch(params: {
  lat: number | null;
  lng: number | null;
  radius_miles?: number;
  service_type?: string;
}) {
  return useQuery<PaginatedResponse<ProviderSearchResult>>({
    queryKey: ["providers", "search", params],
    queryFn: () =>
      apiClient.providers.search({
        lat: params.lat!,
        lng: params.lng!,
        radius_miles: params.radius_miles ?? 10,
        service_type: params.service_type,
      }),
    enabled: params.lat !== null && params.lng !== null,
    staleTime: 30_000,
  });
}

export function useProviderProfile(providerId: string | undefined) {
  return useQuery<ProviderProfile>({
    queryKey: ["providers", providerId],
    queryFn: () => apiClient.providers.getProfile(providerId!),
    enabled: !!providerId,
    staleTime: 60_000,
  });
}

export function useProviderAvailability(providerId: string | undefined) {
  return useQuery<Availability[]>({
    queryKey: ["providers", providerId, "availability"],
    queryFn: () => apiClient.providers.getAvailability(providerId!),
    enabled: !!providerId,
  });
}
```

```typescript
import { useQuery, useQueryClient } from "@tanstack/react-query";
import type { User } from "@mysitter/types";
import { apiClient, clearAccessToken, getAccessTokenSync } from "../lib/apiClient";

export function useAuth() {
  const qc = useQueryClient();
  const { data: user, isLoading } = useQuery<User | null>({
    queryKey: ["auth", "me"],
    queryFn: async () => {
      if (!getAccessTokenSync()) return null;
      return apiClient.users.me();
    },
    staleTime: 5 * 60_000,
    retry: false,
  });

  async function logout() {
    await clearAccessToken();
    qc.setQueryData(["auth", "me"], null);
    qc.clear();
  }

  return { user: user ?? null, isLoading, logout };
}
```

**`apps/api/app/core/storage.py` — repository pattern, S3-compatible:**

```python
"""Object storage abstraction backed by DigitalOcean Spaces (S3-compatible).

All callers go through StorageRepository — the underlying SDK (boto3) is
confined to this file. To swap providers (e.g. AWS S3), only this file changes.
"""
import uuid
from abc import ABC, abstractmethod

import boto3
from botocore.config import Config

from app.core.config import settings

_PRESIGN_EXPIRY_SECONDS = 300  # 5 minutes to complete the upload


class AbstractStorageRepository(ABC):
    @abstractmethod
    def generate_upload_url(self, key: str, content_type: str) -> str:
        raise NotImplementedError("generate_upload_url not implemented")

    @abstractmethod
    def public_url(self, key: str) -> str:
        raise NotImplementedError("public_url not implemented")

    @abstractmethod
    def delete(self, key: str) -> None:
        raise NotImplementedError("delete not implemented")


class SpacesStorageRepository(AbstractStorageRepository):
    """DigitalOcean Spaces via the S3-compatible API."""

    def __init__(self) -> None:
        self._client = boto3.client(
            "s3",
            region_name=settings.spaces_region,
            endpoint_url=settings.spaces_endpoint,
            aws_access_key_id=settings.spaces_key,
            aws_secret_access_key=settings.spaces_secret,
            config=Config(signature_version="s3v4"),
        )
        self._bucket = settings.spaces_bucket

    def generate_upload_url(self, key: str, content_type: str) -> str:
        """Return a presigned PUT URL valid for PRESIGN_EXPIRY_SECONDS."""
        return self._client.generate_presigned_url(
            "put_object",
            Params={
                "Bucket": self._bucket,
                "Key": key,
                "ContentType": content_type,
                "ACL": "public-read",
            },
            ExpiresIn=_PRESIGN_EXPIRY_SECONDS,
        )

    def public_url(self, key: str) -> str:
        endpoint = settings.spaces_endpoint.rstrip("/")
        return f"{endpoint}/{self._bucket}/{key}"

    def delete(self, key: str) -> None:
        self._client.delete_object(Bucket=self._bucket, Key=key)


def get_storage() -> AbstractStorageRepository:
    """FastAPI dependency: return the configured storage repository."""
    return SpacesStorageRepository()
```

**`apps/api/app/routers/uploads.py`:**
```python
"""
Two-phase upload flow:
  1. POST /uploads/presign  → returns a presigned PUT URL + public_url
  2. Client PUTs file directly to Spaces (no server bandwidth used)
  3. Client calls PATCH /users/me or PATCH /providers/me with the returned public_url

There is no server-side confirm step — the presigned URL's ACL is public-read,
so the public_url is valid immediately after a successful PUT.
"""
import uuid

from fastapi import APIRouter, Depends, HTTPException, status

from app.core.deps import get_current_user
from app.core.storage import AbstractStorageRepository, get_storage
from app.models.user import User
from app.schemas.uploads import UploadUrlRequest, UploadUrlResponse

router = APIRouter(prefix="/uploads", tags=["uploads"])

_ALLOWED_CONTENT_TYPES = {"image/jpeg", "image/png", "image/webp", "image/gif"}
_ALLOWED_PURPOSES = {"avatar", "provider_photo"}
_MAX_FILE_SIZE_HINT = 5 * 1024 * 1024  # 5 MB — enforced on the client side only


@router.post("/presign", response_model=UploadUrlResponse, status_code=201)
async def get_presigned_upload_url(
    body: UploadUrlRequest,
    current_user: User = Depends(get_current_user),
    storage: AbstractStorageRepository = Depends(get_storage),
) -> UploadUrlResponse:
    if body.content_type not in _ALLOWED_CONTENT_TYPES:
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            detail=f"content_type must be one of {sorted(_ALLOWED_CONTENT_TYPES)}",
        )
    if body.purpose not in _ALLOWED_PURPOSES:
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            detail=f"purpose must be one of {sorted(_ALLOWED_PURPOSES)}",
        )

    ext = body.content_type.split("/")[1].replace("jpeg", "jpg")
    key = f"{body.purpose}s/{current_user.id}/{uuid.uuid4()}.{ext}"

    upload_url = storage.generate_upload_url(key, body.content_type)
    public_url = storage.public_url(key)

    return UploadUrlResponse(upload_url=upload_url, public_url=public_url, key=key)
```

```typescript
import { useMutation } from "@tanstack/react-query";
import { apiClient } from "@/lib/apiClient";
import type { UploadUrlRequest } from "@mysitter/types";

export function useUpload() {
  return useMutation({
    mutationFn: async ({
      file,
      purpose,
    }: {
      file: File;
      purpose: UploadUrlRequest["purpose"];
    }) => {
      // Step 1: get presigned URL
      const { upload_url, public_url } = await apiClient.uploads.presign({
        content_type: file.type,
        purpose,
      });

      // Step 2: PUT directly to Spaces — no auth header, presigned URL handles it
      const res = await fetch(upload_url, {
        method: "PUT",
        body: file,
        headers: { "Content-Type": file.type },
      });
      if (!res.ok) throw new Error(`Upload failed: HTTP ${res.status}`);

      // Return the permanent public URL for the caller to persist
      return public_url;
    },
  });
}
```

```python
"""
Email rendering via Jinja2 HTML templates.

All templates live in apps/api/app/email_templates/.
Call EmailRenderer.render(template_name, context) to get the final HTML string.
The renderer is intentionally a plain class (no FastAPI dependency injection)
because it is called from synchronous Celery workers.
"""
from pathlib import Path

from jinja2 import Environment, FileSystemLoader, select_autoescape


_TEMPLATE_DIR = Path(__file__).parent.parent / "email_templates"


class EmailRenderer:
    _env: Environment | None = None

    @classmethod
    def _get_env(cls) -> Environment:
        if cls._env is None:
            cls._env = Environment(
                loader=FileSystemLoader(str(_TEMPLATE_DIR)),
                autoescape=select_autoescape(["html"]),
            )
        return cls._env

    @classmethod
    def render(cls, template_name: str, context: dict) -> str:
        """Render a template file with the given context dict."""
        template = cls._get_env().get_template(template_name)
        return template.render(**context)
```

```tsx
import { useEffect } from "react";

interface SeoHeadProps {
  title: string;
  description: string;
  ogImage?: string | null;
  canonicalUrl?: string;
  jsonLd?: Record<string, unknown>;
}

export function SeoHead({ title, description, ogImage, canonicalUrl, jsonLd }: SeoHeadProps) {
  useEffect(() => {
    document.title = title;

    const setMeta = (name: string, content: string, prop = false) => {
      const attr = prop ? "property" : "name";
      let el = document.querySelector<HTMLMetaElement>(`meta[${attr}="${name}"]`);
      if (!el) {
        el = document.createElement("meta");
        el.setAttribute(attr, name);
        document.head.appendChild(el);
      }
      el.content = content;
    };

    setMeta("description", description);
    setMeta("og:title", title, true);
    setMeta("og:description", description, true);
    if (ogImage) setMeta("og:image", ogImage, true);
    if (canonicalUrl) setMeta("og:url", canonicalUrl, true);

    if (jsonLd) {
      let script = document.querySelector<HTMLScriptElement>('script[type="application/ld+json"]');
      if (!script) {
        script = document.createElement("script");
        script.type = "application/ld+json";
        document.head.appendChild(script);
      }
      script.textContent = JSON.stringify(jsonLd);
    }

    return () => {
      document.title = "mysitter";
    };
  }, [title, description, ogImage, canonicalUrl, jsonLd]);

  return null;
}
```

```python
import secrets
import uuid
from datetime import datetime, timezone

from fastapi import HTTPException, status
from sqlalchemy import select, func
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.config import settings
from app.core.email import EmailRenderer
from app.models.referral import Referral, ReferralStatus
from app.models.user import User
from app.schemas.referral import ReferralResponse, ReferralStatsResponse


class ReferralService:
    def __init__(self, db: AsyncSession) -> None:
        self._db = db

    async def get_or_create_referral_code(self, user: User) -> Referral:
        """Return the user's existing (unused) referral entry, or create one."""
        existing = await self._db.scalar(
            select(Referral)
            .where(Referral.referrer_id == user.id)
            .where(Referral.invitee_id.is_(None))
            .order_by(Referral.created_at.desc())
        )
        if existing:
            return existing

        code = secrets.token_urlsafe(10)[:10].upper()  # 10-char alphanumeric code
        referral = Referral(referrer_id=user.id, code=code)
        self._db.add(referral)
        await self._db.commit()
        await self._db.refresh(referral)
        return referral

    def _invite_url(self, code: str) -> str:
        return f"{settings.webauthn_origin}/register?ref={code}"

    async def get_stats(self, user: User) -> ReferralStatsResponse:
        total_sent = await self._db.scalar(
            select(func.count(Referral.id)).where(Referral.referrer_id == user.id)
        ) or 0
        total_converted = await self._db.scalar(
            select(func.count(Referral.id))
            .where(Referral.referrer_id == user.id)
            .where(Referral.status == ReferralStatus.CONVERTED)
        ) or 0
        pending = await self._db.scalar(
            select(func.count(Referral.id))
            .where(Referral.referrer_id == user.id)
            .where(Referral.status == ReferralStatus.PENDING)
            .where(Referral.invitee_id.isnot(None))
        ) or 0

        referral = await self.get_or_create_referral_code(user)
        return ReferralStatsResponse(
            total_sent=total_sent,
            total_converted=total_converted,
            pending=pending,
            invite_url=self._invite_url(referral.code),
        )

    async def send_invite_email(self, referrer: User, invitee_email: str) -> None:
        """Send a branded invite email. Import resend inline to avoid module-level side effects."""
        referral = await self.get_or_create_referral_code(referrer)
        invite_url = self._invite_url(referral.code)

        if not settings.resend_api_key:
            return  # no-op in dev/test

        import resend
        resend.api_key = settings.resend_api_key
        resend.Emails.send({
            "from": settings.email_from,
            "to": [invitee_email],
            "subject": f"{referrer.display_name} invited you to mysitter",
            "html": EmailRenderer.render("referral_invite.html", {
                "referrer_name": referrer.display_name,
                "invite_url": invite_url,
                "app_url": settings.webauthn_origin,
            }),
        })

    async def redeem_code(self, code: str, invitee: User) -> None:
        """
        Called during registration when ?ref=<code> is present.
        Links the invitee to the referral row and sets referred_by_id.
        """
        referral = await self._db.scalar(
            select(Referral).where(Referral.code == code)
        )
        if not referral or referral.invitee_id is not None:
            return  # expired or already claimed — silently ignore

        if referral.referrer_id == invitee.id:
            return  # self-referral — silently ignore

        referral.invitee_id = invitee.id
        referral.status = ReferralStatus.PENDING
        invitee.referred_by_id = referral.referrer_id
        await self._db.commit()

    async def mark_converted(self, invitee_id: uuid.UUID) -> None:
        """
        Called from StripeService when the invitee activates a paid subscription.
        Marks the referral CONVERTED. The reward dispatch is deferred to Phase 5.
        """
        referral = await self._db.scalar(
            select(Referral)
            .where(Referral.invitee_id == invitee_id)
            .where(Referral.status == ReferralStatus.PENDING)
        )
        if not referral:
            return

        referral.status = ReferralStatus.CONVERTED
        referral.converted_at = datetime.now(timezone.utc)
        await self._db.commit()
```

```python
"""
Post-deploy smoke test. Run after each deployment to verify the new
instance is healthy before cutting over traffic.

Usage: python deploy/healthcheck.py --host https://api.mysitter.com
"""
import argparse
import sys
import time

import requests


def check(host: str) -> bool:
    endpoints = [
        ("/health", 200),
        ("/providers/search?lat=40.71&lng=-74.01&radius_miles=5", 200),
    ]
    for path, expected_status in endpoints:
        try:
            resp = requests.get(f"{host}{path}", timeout=10)
            if resp.status_code != expected_status:
                print(f"FAIL {path}: got {resp.status_code}, expected {expected_status}")
                return False
            print(f"OK   {path}")
        except requests.RequestException as exc:
            print(f"FAIL {path}: {exc}")
            return False
    return True


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--host", required=True)
    parser.add_argument("--retries", type=int, default=5)
    parser.add_argument("--delay", type=int, default=5)
    args = parser.parse_args()

    for attempt in range(1, args.retries + 1):
        print(f"Attempt {attempt}/{args.retries}...")
        if check(args.host):
            print("Healthcheck passed.")
            sys.exit(0)
        if attempt < args.retries:
            time.sleep(args.delay)

    print("Healthcheck failed after all retries.")
    sys.exit(1)
```

## Key Architecture Decisions — Do Not Second-Guess These

1. **Image uploads use presigned PUT URLs — the file never transits the API server.** The client gets a presigned URL from `POST /uploads/presign`, then PUTs the file directly to DigitalOcean Spaces. The API server never buffers binary file data. This is the only approach that scales — a 5 MB avatar through 4 Uvicorn workers would saturate a 2 GB droplet quickly. There is no server-side confirm step; `ACL: public-read` means the URL is valid immediately.

2. **Referral reward dispatch is explicitly deferred.** `ReferralStatus.CONVERTED` is set when the Stripe `invoice.paid` webhook fires for the invitee's first paid subscription. The actual reward (credit, discount coupon) is a Phase 5 concern that requires a credit balance model and potentially Stripe Coupon API. The `REWARDED` enum value is defined now so Phase 5 has a clean state transition to implement — no placeholder columns or half-implemented logic.

3. **SEO meta is served from a separate `/seo/` endpoint, not embedded in the main provider endpoint.** The main `GET /providers/{id}` response is used by the booking flow and needs to stay lean (no 160-character description generation on every profile fetch). SEO data is fetched once per page load with a 10-minute `staleTime` and has its own cache path. Adding `json_ld` to `ProviderProfileResponse` would bloat every API call that references a provider.

4. **The `slug` backfill runs in the Alembic migration as raw SQL.** It must be idempotent (only updates rows where `slug IS NULL`) and execute before the `NOT NULL` constraint is applied. Do not set `nullable=False` on the slug column in the same migration that backfills it — run the backfill, verify, then add a separate `NOT NULL` migration.

5. **Read replica sessions are dependency-injected via `get_read_db`.** Services that are called from both write (booking creation) and read (search) paths do not know which session they receive — they just accept an `AsyncSession`. The router decides whether to inject `get_db` (write path) or `get_read_db` (read path). Never add replica-awareness inside a service class.

6. **The mobile app uses the same `packages/api-client` as the web app.** New API namespaces added to `ApiClient` (referrals, uploads, SEO) are immediately available to mobile via `apiClient.referrals.getMyStats()` etc. No mobile-specific API wrapper is needed. The only mobile-specific code is the `expo-secure-store` token adapter in `apps/mobile/lib/apiClient.ts`.

7. **Referral self-referrals and re-claimed codes are silently ignored.** `ReferralService.redeem_code()` returns without error in both cases. This avoids 400 errors on legitimate race conditions (e.g., user registers with an expired link) and prevents information leakage about which codes are valid. The referrer sees no conversion — that is the correct outcome.

8. **The `celery_beat` service is separated from `celery_worker` in the production compose.** Celery Beat must run as exactly one process cluster-wide (it is not horizontally scalable — it would send duplicate scheduled tasks). In `docker-compose.prod.yml`, `celery_beat` is a distinct service so it can be pinned to a single node or managed separately if scaling workers later.
